# App de posts (Tech Challenge 4): plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** App Expo que lista, busca, mostra, cria, edita e exclui posts educacionais da API do Tech Challenge 3, no visual "Vibrante".

**Architecture:** Expo Router com um Stack (telas sem header nativo, login como modal). A camada `src/api` só conhece HTTP; `src/auth` guarda a sessão (secure-store / localStorage) e expõe um contexto; `src/lib` tem as regras puras, cobertas por testes; telas só compõem hooks e componentes. Estilo com `StyleSheet` e tokens de `src/theme.ts`.

**Tech Stack:** Expo SDK 57, React Native 0.86, React 19.2, Expo Router 57, TypeScript 6, expo-secure-store, expo-font + @expo-google-fonts, Jest via jest-expo.

**Spec:** `docs/superpowers/specs/2026-11-02-app-posts-design.md`

## Review Focus

1. Busca com caracteres especiais (`c++`, `(`, `.`, `*`) não pode quebrar o destaque do termo: o texto aparece normal e o termo é destacado literalmente. Teste em Task 3 (`highlight.test.ts`).
2. Digitar rápido dispara buscas que voltam fora de ordem: só a resposta da busca mais recente atualiza a lista. Teste em Task 3 (`latest.test.ts`).
3. Sessão guardada corrompida, incompleta ou vencida: o app abre deslogado, sem travar. Teste em Task 5 (`session.test.ts`).
4. API devolve corpo não-JSON (página HTML de erro 500 do Next) ou fica fora do ar: vira `ApiError` com mensagem legível, sem exceção solta. Teste em Task 4 (`client.test.ts`).
5. Busca só com espaços não manda `q` vazio, e post sem autor (`authors` nulo) mostra "Autor desconhecido" em vez de quebrar. Testes em Task 4 (`client.test.ts`) e Task 3 (`post.test.ts`).

```ts
export function excerpt(text: string, max = 160): string {
  const clean = text.replace(/\s+/g, ' ').trim();
  if (clean.length <= max) return clean;
  const cut = clean.slice(0, max);
  const lastSpace = cut.lastIndexOf(' ');
  return `${(lastSpace > 0 ? cut.slice(0, lastSpace) : cut).trimEnd()}…`;
}
```

```ts
export function formatDate(iso: string): string {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return '';
  return `${d.getDate()} ${MONTHS[d.getMonth()]} ${d.getFullYear()}`;
}
```

```ts
export type HighlightPart = { text: string; match: boolean };

function escapeRegExp(s: string) {
  return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

export function splitHighlight(text: string, term: string): HighlightPart[] {
  const t = term.trim();
  if (!t) return [{ text, match: false }];
  const re = new RegExp(`(${escapeRegExp(t)})`, 'gi');
  return text
    .split(re)
    .filter((part) => part !== '')
    .map((part) => ({ text: part, match: part.toLowerCase() === t.toLowerCase() }));
}
```

```ts
export function initials(name: string | null | undefined): string {
  const words = (name ?? '').trim().split(/\s+/).filter(Boolean);
  if (words.length === 0) return '?';
  const first = words[0][0];
  const last = words.length > 1 ? words[words.length - 1][0] : '';
  return (first + last).toUpperCase();
}
```

`src/lib/latest.ts`:

```ts
export function createLatest() {
  let current = 0;
  return async function latest<T>(promise: Promise<T>): Promise<{ stale: boolean; value: T }> {
    const id = ++current;
    const value = await promise;
    return { stale: id !== current, value };
  };
}
```

```ts
import type { Post, SessionUser } from '@/api/types';

export function authorName(post: Pick<Post, 'authors'>): string {
  const name = post.authors?.name?.trim();
  return name ? name : 'Autor desconhecido';
}

export function canCreatePost(user: SessionUser | null): boolean {
  return user?.role === 'AUTHOR' || user?.role === 'ADMIN';
}

export function canManagePost(user: SessionUser | null, post: Pick<Post, 'authorId'>): boolean {
  if (!user) return false;
  if (user.role === 'ADMIN') return true;
  return user.authorId !== null && user.authorId === post.authorId;
}
```

```ts
export class ApiError extends Error {
  status: number;
  fieldErrors?: Record<string, string[]>;

  constructor(status: number, message: string, fieldErrors?: Record<string, string[]>) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
    this.fieldErrors = fieldErrors;
  }
}

type RequestOptions = {
  method?: 'GET' | 'POST' | 'PUT' | 'DELETE';
  body?: unknown;
  token?: string | null;
  query?: Record<string, string | undefined>;
};

function getApiUrl(): string {
  // Acesso literal: o Expo só embute EXPO_PUBLIC_* escrito assim.
  const url = process.env.EXPO_PUBLIC_API_URL;
  if (!url) {
    throw new ApiError(0, 'Endereço da API não configurado (EXPO_PUBLIC_API_URL no .env).');
  }
  return url.replace(/\/+$/, '');
}

export async function apiRequest<T>(path: string, opts: RequestOptions = {}): Promise<T> {
  const params = new URLSearchParams();
  for (const [key, value] of Object.entries(opts.query ?? {})) {
    const v = value?.trim();
    if (v) params.set(key, v);
  }
  const qs = params.toString();
  const url = `${getApiUrl()}${path}${qs ? `?${qs}` : ''}`;

  const headers: Record<string, string> = { Accept: 'application/json' };
  if (opts.body !== undefined) headers['Content-Type'] = 'application/json';
  if (opts.token) headers.Authorization = `Bearer ${opts.token}`;

  let res: Response;
  try {
    res = await fetch(url, {
      method: opts.method ?? 'GET',
      headers,
      body: opts.body !== undefined ? JSON.stringify(opts.body) : undefined,
      // A API também define cookie de sessão; o app usa só o token.
      credentials: 'omit',
    });
  } catch {
    throw new ApiError(0, 'Sem conexão com o servidor.');
  }

  const data: unknown = await res.json().catch(() => null);

  if (!res.ok) {
    const obj = (data && typeof data === 'object' ? data : {}) as {
      error?: unknown;
      errors?: Record<string, string[]>;
    };
    const fieldErrors = obj.errors && typeof obj.errors === 'object' ? obj.errors : undefined;
    const message =
      typeof obj.error === 'string' && obj.error
        ? obj.error
        : fieldErrors
          ? 'Confira os campos destacados.'
          : res.status >= 500
            ? `Erro ${res.status} no servidor.`
            : `Erro ${res.status}.`;
    throw new ApiError(res.status, message, fieldErrors);
  }

  return data as T;
}
```

```ts
import type { Role, SessionUser } from '@/api/types';

export type Session = { token: string; expiresAt: string; user: SessionUser };

const ROLES: Role[] = ['ADMIN', 'AUTHOR', 'STUDENT'];

export function parseStoredSession(raw: string | null, now: Date = new Date()): Session | null {
  if (!raw) return null;
  let data: unknown;
  try {
    data = JSON.parse(raw);
  } catch {
    return null;
  }
  if (!data || typeof data !== 'object') return null;
  const s = data as Partial<Session>;
  const u = s.user as Partial<SessionUser> | undefined;
  const expires = typeof s.expiresAt === 'string' ? new Date(s.expiresAt) : null;

  const ok =
    typeof s.token === 'string' && s.token.length > 0 &&
    expires !== null && !Number.isNaN(expires.getTime()) && expires > now &&
    !!u && typeof u.id === 'string' && typeof u.name === 'string' &&
    ROLES.includes(u.role as Role) &&
    (u.authorId === null || typeof u.authorId === 'string');

  return ok ? (s as Session) : null;
}
```

```tsx
import { createContext, useCallback, useContext, useEffect, useMemo, useState, type ReactNode } from 'react';

import { login } from '@/api/auth';
import type { SessionUser } from '@/api/types';
import { parseStoredSession, type Session } from './session';
import { clearRaw, loadRaw, saveRaw } from './tokenStorage';

type AuthValue = {
  ready: boolean;
  session: Session | null;
  user: SessionUser | null;
  token: string | null;
  signIn(email: string, password: string): Promise<void>;
  signOut(): Promise<void>;
};

const AuthContext = createContext<AuthValue | null>(null);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [ready, setReady] = useState(false);
  const [session, setSession] = useState<Session | null>(null);

  useEffect(() => {
    let alive = true;
    loadRaw()
      .catch(() => null)
      .then((raw) => {
        if (!alive) return;
        const parsed = parseStoredSession(raw);
        setSession(parsed);
        if (raw && !parsed) clearRaw().catch(() => {});
        setReady(true);
      });
    return () => {
      alive = false;
    };
  }, []);

  const signIn = useCallback(async (email: string, password: string) => {
    const res = await login(email, password);
    const next: Session = { token: res.token, expiresAt: res.expiresAt, user: res.user };
    setSession(next);
    await saveRaw(JSON.stringify(next)).catch(() => {});
  }, []);

  const signOut = useCallback(async () => {
    setSession(null);
    await clearRaw().catch(() => {});
  }, []);

  const value = useMemo<AuthValue>(
    () => ({
      ready,
      session,
      user: session?.user ?? null,
      token: session?.token ?? null,
      signIn,
      signOut,
    }),
    [ready, session, signIn, signOut],
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth(): AuthValue {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth precisa estar dentro de <AuthProvider>.');
  return ctx;
}
```

```ts
import { useEffect, useState } from 'react';

export function useDebouncedValue<T>(value: T, delay: number): T {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(t);
  }, [value, delay]);
  return debounced;
}
```

```ts
// src/hooks/usePosts.ts
import { useCallback, useRef, useState } from 'react';

import { ApiError } from '@/api/client';
import { listPosts } from '@/api/posts';
import type { Post } from '@/api/types';
import { createLatest } from '@/lib/latest';

type Status = 'loading' | 'success' | 'error';

export function usePosts(query: string) {
  const latest = useRef(createLatest()).current;
  const [status, setStatus] = useState<Status>('loading');
  const [posts, setPosts] = useState<Post[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [refreshing, setRefreshing] = useState(false);

  const reload = useCallback(
    (opts?: { pull?: boolean }) => {
      if (opts?.pull) setRefreshing(true);
      else setStatus((s) => (s === 'success' ? s : 'loading'));
      latest(listPosts(query))
        .then(({ stale, value }) => {
          if (stale) return;
          setPosts(value);
          setError(null);
          setStatus('success');
        })
        .catch((err: unknown) => {
          setError(err instanceof ApiError ? err.message : 'Algo deu errado.');
          setStatus('error');
        })
        .finally(() => setRefreshing(false));
    },
    [latest, query],
  );

  return { status, posts, error, refreshing, reload };
}
```

Obs.: um erro de uma busca antiga pode sobrescrever o resultado de uma mais nova; é aceitável (mostra "Tentar de novo"), e o `createLatest` já protege o caso comum (respostas de sucesso fora de ordem).

```tsx
import { deletePost } from '@/api/posts';
import { useAuth } from '@/auth/AuthContext';
import { CategoryTag } from '@/components/CategoryTag';
import { ConfirmSheet } from '@/components/ConfirmSheet';
import { IconButton } from '@/components/IconButton';
import { StateMessage } from '@/components/StateMessage';
import { useHandleApiError } from '@/hooks/useAuthGuard';
import { usePost } from '@/hooks/usePost';
import { formatDate } from '@/lib/formatDate';
import { authorName, canManagePost } from '@/lib/post';
import { colors, fonts, getCategory, radius, space } from '@/theme';

export default function PostScreen() {
  const { id } = useLocalSearchParams<{ id: string }>();
  const { user, token } = useAuth();
  const handleApiError = useHandleApiError();
  const { status, post, error, reload } = usePost(id);
  const [confirming, setConfirming] = useState(false);
  const [deleting, setDeleting] = useState(false);

  useFocusEffect(useCallback(() => reload(), [reload]));

  function back() {
    if (router.canGoBack()) router.back();
    else router.replace('/');
  }

  async function confirmDelete() {
    if (!post || !token) return;
    setDeleting(true);
    try {
      await deletePost(post.id, token);
      setConfirming(false);
      router.dismissTo({ pathname: '/', params: { aviso: 'Post excluído' } });
    } catch (err) {
      setConfirming(false);
      if (!handleApiError(err)) {
        Alert.alert('Não foi possível excluir', err instanceof ApiError ? err.message : 'Tente de novo.');
      }
    } finally {
      setDeleting(false);
    }
  }

  if (status === 'loading' && !post) {
    return (
      <SafeAreaView style={[styles.screen, styles.center]}>
        <ActivityIndicator color={colors.text} />
      </SafeAreaView>
    );
  }

  if (!post) {
    const notFound = error?.status === 404;
    return (
      <SafeAreaView style={styles.screen}>
        <View style={styles.nav}><IconButton label="←" onPress={back} accessibilityLabel="Voltar" /></View>
        <StateMessage
          emoji={notFound ? '🔎' : '📡'}
          title={notFound ? 'Post não encontrado' : 'Sem conexão com o servidor'}
          message={notFound ? 'Ele pode ter sido excluído.' : error?.message}
          actionLabel={notFound ? 'Voltar' : 'Tentar de novo'}
          onAction={notFound ? back : reload}
        />
      </SafeAreaView>
    );
  }

  const cat = getCategory(post.category);
  const canManage = canManagePost(user, post);

  return (
    <SafeAreaView style={styles.screen} edges={['top']}>
      <ScrollView contentContainerStyle={styles.content}>
        <View style={styles.nav}>
          <IconButton label="←" onPress={back} accessibilityLabel="Voltar" />
          {canManage && (
            <View style={styles.actions}>
              <IconButton label="✎ Editar" accessibilityLabel="Editar post"
                onPress={() => router.push({ pathname: '/post/[id]/editar', params: { id: post.id } })} />
              <IconButton label="🗑" tone="danger" accessibilityLabel="Excluir post" onPress={() => setConfirming(true)} />
            </View>
          )}
        </View>
        <View style={[styles.hero, { backgroundColor: cat.color }]}>
          <CategoryTag category={post.category} />
          <Text style={styles.title} accessibilityRole="header">{post.title}</Text>
          <Text style={styles.by}>{authorName(post)} · {formatDate(post.createdAt)}</Text>
        </View>
        <Text style={styles.body} selectable>{post.text}</Text>
      </ScrollView>
      <ConfirmSheet
        visible={confirming}
        title="Excluir este post?"
        message={`“${post.title}” vai sumir para todos os alunos. Não dá para desfazer.`}
        confirmLabel="Excluir post"
        onConfirm={confirmDelete}
        onCancel={() => setConfirming(false)}
        loading={deleting}
      />
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, backgroundColor: colors.background },
  center: { alignItems: 'center', justifyContent: 'center' },
  content: { padding: space.lg, paddingBottom: 40 },
  nav: { flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 14, paddingHorizontal: 0 },
  actions: { flexDirection: 'row', gap: 8 },
  hero: { borderRadius: radius.hero, padding: 18, marginBottom: 16 },
  title: { fontFamily: fonts.display800, fontSize: 26, lineHeight: 28, color: colors.onCard, marginBottom: 10 },
  by: { fontFamily: fonts.body700, fontSize: 12, color: colors.onCard },
  body: { fontFamily: fonts.body400, fontSize: 14, lineHeight: 22, color: 'rgba(255,255,255,0.88)' },
});
```

- [ ] **Step 2: `PostForm.tsx`**

```tsx
import { useState } from 'react';
import { KeyboardAvoidingView, Platform, ScrollView, StyleSheet, Text, View } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';

import { ApiError } from '@/api/client';
import type { PostInput } from '@/api/types';
import { colors, fonts, space } from '@/theme';
import { CategoryChips } from './CategoryChips';
import { Field } from './Field';
import { IconButton } from './IconButton';
import { PrimaryButton } from './PrimaryButton';

type Props = {
  title: string;
  submitLabel: string;
  initial: PostInput;
  onSubmit: (input: PostInput) => Promise<void>;
  onClose: () => void;
};

export function PostForm({ title, submitLabel, initial, onSubmit, onClose }: Props) {
  const [values, setValues] = useState<PostInput>(initial);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [fieldErrors, setFieldErrors] = useState<Record<string, string[]>>({});

  async function submit() {
    setSaving(true);
    setError(null);
    setFieldErrors({});
    try {
      await onSubmit({ ...values, title: values.title.trim(), text: values.text.trim() });
    } catch (err) {
      if (err instanceof ApiError) {
        setFieldErrors(err.fieldErrors ?? {});
        if (!err.fieldErrors) setError(err.message);
      } else {
        setError('Algo deu errado. Tente de novo.');
      }
    } finally {
      setSaving(false);
    }
  }

  return (
    <SafeAreaView style={styles.screen}>
      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : undefined}>
        <ScrollView contentContainerStyle={styles.content} keyboardShouldPersistTaps="handled">
          <View style={styles.nav}>
            <IconButton label="✕" onPress={onClose} accessibilityLabel="Fechar" />
            <Text style={styles.navTitle} accessibilityRole="header">{title}</Text>
            <View style={{ width: 40 }} />
          </View>
          {error ? <Text style={styles.error} accessibilityRole="alert">{error}</Text> : null}
          <Field label="Título" value={values.title} onChangeText={(t) => setValues((v) => ({ ...v, title: t }))}
            error={fieldErrors.title?.[0]} maxLength={140} />
          <Text style={styles.label}>Matéria</Text>
          <CategoryChips value={values.category} onChange={(c) => setValues((v) => ({ ...v, category: c }))} />
          {fieldErrors.category?.[0] ? <Text style={styles.error}>{fieldErrors.category[0]}</Text> : null}
          <Field label="Conteúdo" value={values.text} onChangeText={(t) => setValues((v) => ({ ...v, text: t }))}
            multiline error={fieldErrors.text?.[0]} />
          <PrimaryButton label={submitLabel} onPress={submit} loading={saving} />
        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, backgroundColor: colors.background },
  content: { padding: space.lg, paddingBottom: 40 },
  nav: { flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between', marginBottom: 18 },
  navTitle: { fontFamily: fonts.display700, fontSize: 16, color: colors.text },
  label: { fontFamily: fonts.body700, fontSize: 12, color: colors.textMuted, marginBottom: 6 },
  error: { fontFamily: fonts.body500, fontSize: 13, color: colors.danger, marginBottom: 12 },
});
```

```tsx
import { router } from 'expo-router';
import { useEffect } from 'react';

import { createPost } from '@/api/posts';
import type { PostInput } from '@/api/types';
import { useAuth } from '@/auth/AuthContext';
import { PostForm } from '@/components/PostForm';
import { useHandleApiError } from '@/hooks/useAuthGuard';
import { canCreatePost } from '@/lib/post';

const EMPTY: PostInput = { title: '', text: '', category: 'GERAL' };

export default function NewPost() {
  const { user, token } = useAuth();
  const handleApiError = useHandleApiError();

  useEffect(() => {
    if (!canCreatePost(user)) router.replace('/login');
  }, [user]);

  async function submit(input: PostInput) {
    if (!token) return router.replace('/login');
    try {
      await createPost(input, token);
      router.back();
    } catch (err) {
      if (!handleApiError(err)) throw err;
    }
  }

  return (
    <PostForm title="Novo post" submitLabel="Publicar" initial={EMPTY} onSubmit={submit} onClose={() => router.back()} />
  );
}
```

```tsx
import { router, useLocalSearchParams } from 'expo-router';
import { useEffect } from 'react';
import { ActivityIndicator, View } from 'react-native';

import { updatePost } from '@/api/posts';
import type { PostInput } from '@/api/types';
import { useAuth } from '@/auth/AuthContext';
import { PostForm } from '@/components/PostForm';
import { StateMessage } from '@/components/StateMessage';
import { useHandleApiError } from '@/hooks/useAuthGuard';
import { usePost } from '@/hooks/usePost';
import { canManagePost } from '@/lib/post';
import { colors, getCategory } from '@/theme';

export default function EditPost() {
  const { id } = useLocalSearchParams<{ id: string }>();
  const { user, token } = useAuth();
  const handleApiError = useHandleApiError();
  const { status, post, error, reload } = usePost(id);

  useEffect(() => reload(), [reload]);

  if (status === 'loading') {
    return (
      <View style={{ flex: 1, backgroundColor: colors.background, alignItems: 'center', justifyContent: 'center' }}>
        <ActivityIndicator color={colors.text} />
      </View>
    );
  }

  if (!post || !canManagePost(user, post)) {
    return (
      <View style={{ flex: 1, backgroundColor: colors.background }}>
        <StateMessage
          emoji="🔒"
          title={post ? 'Você não pode editar este post' : 'Post não encontrado'}
          message={post ? undefined : error?.message}
          actionLabel="Voltar"
          onAction={() => router.back()}
        />
      </View>
    );
  }

  async function submit(input: PostInput) {
    if (!token) return router.replace('/login');
    try {
      await updatePost(id, input, token);
      router.back();
    } catch (err) {
      if (!handleApiError(err)) throw err;
    }
  }

  return (
    <PostForm
      title="Editar post"
      submitLabel="Salvar"
      initial={{ title: post.title, text: post.text, category: getCategory(post.category).value }}
      onSubmit={submit}
      onClose={() => router.back()}
    />
  );
}
```
